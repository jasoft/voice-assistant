import os
import re

def migrate_file(filepath):
    with open(filepath, 'r') as f:
        content = f.read()
    
    # 1. 移除 (unittest.TestCase)
    content = re.sub(r'\(unittest\.TestCase\)', '', content)
    
    # 2. 移除 self.assertEqual (处理多行)
    # 这是一个简化版的处理，通过查找 self.assertEqual( 开头并尝试找到闭合括号
    # 为了安全，我们先处理简单的单行
    # content = re.sub(r'self\.assertEqual\((.*),\s*(.*)\)', r'assert \1 == \2', content)
    
    # 既然有 500 多行，我用更稳的方法：逐个字符查找平衡括号
    def replace_assert(match):
        func_name = match.group(1) # assertEqual, assertIn, etc.
        args_str = match.group(2)
        
        # 这里很难用正则分拆两个参数，因为参数本身可能有逗号
        # 所以我们干脆保留 self.assertEqual 并在类里加一个兼容层？
        # 不，大王要的是纯 pytest。
        return match.group(0)

    # 简单的正则替换，处理那些 self.assertEqual(a, b) 且 a, b 不含嵌套括号的
    content = re.sub(r'self\.assertEqual\(([^,()]*),\s*(.*)\)', r'assert \1 == \2', content)
    content = re.sub(r'self\.assertIn\(([^,()]*),\s*(.*)\)', r'assert \1 in \2', content)
    content = re.sub(r'self\.assertTrue\((.*)\)', r'assert \1', content)
    content = re.sub(r'self\.assertIsNone\((.*)\)', r'assert \1 is None', content)
    content = re.sub(r'self\.assertIsNotNone\((.*)\)', r'assert \1 is not None', content)
    
    # 移除 self.
    # 注意：self.bb, self.config 这种不能删！
    # 只有 self.assertEqual 这种方法调用才需要改。
    # 我们已经把 assertEqual 换成了 assert。
    
    with open(filepath, 'w') as f:
        f.write(content)
